/*Crear las tablas*/

-- CREAR TABLA EQUIPOS (antes para poder usar la foreign key)
CREATE TABLE equipos (
nombre VARCHAR (50) PRIMARY KEY,
categoria VARCHAR(50) NOT NULL,
posicion_en_liga INT NOT NULL,
numero_jugadores INT NOT NULL,
ADA VARCHAR(45) NOT NULL);


-- CREAR TABLA DE JUGADORES 
CREATE TABLE jugadores (
nif VARCHAR (9) PRIMARY KEY,
nombre VARCHAR(50) NOT NULL,
apellidos VARCHAR(50) NOT NULL,
edad INT NOT NULL,
partidos_jugados INT NOT NULL,
importe_prima INT NOT NULL,
nom_equipo VARCHAR (50) NOT NULL,
JBC VARCHAR(45) NOT NULL,
FOREIGN KEY (nom_equipo) REFERENCES equipos(nombre));

-- CREAR TABLA TOTALES_TABLA

CREATE TABLE totales_tabla (
    tabla VARCHAR(20) PRIMARY KEY,
    num_registros INT NOT NULL,
    PPE VARCHAR(45) NOT NULL,
    SGA VARCHAR(45) NOT NULL);

/*TRIGGERS*/

--trigger de alta jugadores
drop trigger if exists trigger_alta_jugador;

delimiter $$

create trigger trigger_alta_jugador
after insert on jugadores
for each row
begin
-- Actualizamos el contador en la tabla equipos
update equipos set numero_jugadores = numero_jugadores + 1 where nombre = new.nom_equipo;

-- Actualizamos el contador global en totales_tabla
update totales_tabla set num_registros = num_registros + 1 where tabla = 'jugadores';

end $$
delimiter ;


-- trigger baja de jugadores 

drop trigger if exists trigger_baja_jugadores;
delimiter $$
create trigger trigger_baja_jugadores
after delete on jugadores
for each row
begin

declare num_jugadores int;
declare posicion_equipo int;

-- restar un jugador menos del equipo
update equipos set numero_jugadores = numero_jugadores - 1 where nombre = old.nom_equipo;

-- Restar 1 jugador al total de registros 
update totales_tabla set num_registros = num_registros - 1 where tabla = 'jugadores';

-- apartado de eliminar el equipo si no quedan ningún jugador dentro

-- ver cuantos jugadores quedan en el equipo
select numero_jugadores into num_jugadores from equipos where nombre = old.nom_equipo;

-- si no quedan jugadores borrar el equipo y subir los otros equipos una posición
if num_jugadores = 0 then

	-- guardar la posición del equipo y borrar el equipo
	select posicion_en_liga into posicion_equipo from equipos where nombre = old.nom_equipo;
	delete from equipos where nombre = old.nom_equipo;

	-- Subir posiciones de los equipos por debajo (comparando la variable con la posición guardada y camparandola con las posiciones en liga)
	update equipos set posicion_en_liga = posicion_en_liga - 1 where posicion_en_liga > posicion_equipo;
end if;
end $$

delimiter ;


-- trigger alta de equipo

drop trigger if exists trigger_alta_equipo;
delimiter $$

create trigger trigger_alta_equipo
after insert on equipos 
for each row 
begin 
 -- modifca la tabla, agregamos 1 al num de registros en la seccion de equipos
    update totales_tabla set num_registros = num_registros + 1 where tabla = 'equipos';
end $$
delimiter ;


-- trigger baja de equipo

drop trigger if exists trigger_baja_equipo;
delimiter $$

create trigger trigger_baja_equipo
after delete on equipos
for each row
begin
-- modificamos la tabla restando esta vez uno en la sección de equipos
update totales_tabla set num_registros = num_registros - 1 where tabla = 'equipos';
end $$
delimiter ;

/*VISTAS*/

-- vista de los datos de los jugadores

CREATE VIEW view_datos_equipos AS 
SELECT jugadores.apellidos, jugadores.nombre,jugadores.nif,jugadores.edad,equipos.nombre AS nombre_equipo, equipos.categoria, equipos.posicion_en_liga FROM jugadores
INNER JOIN equipos ON jugadores.nom_equipo = equipos.nombre ORDER BY jugadores.apellidos ASC, jugadores.nombre ASC;

-- vista de descuadre
create view view_descuadre as
select equipos.* from equipos where numero_jugadores != (select count(*) from jugadores where jugadores.nom_equipo = equipos.nombre);
