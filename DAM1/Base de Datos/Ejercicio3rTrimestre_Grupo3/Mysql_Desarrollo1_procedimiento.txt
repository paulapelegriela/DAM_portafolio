-- procedimiento para calcular la letra del dni
drop procedure if exists proc_calcular_letra_nif;
delimiter $$

create procedure proc_calcular_letra_nif()

begin

declare wdni int;
declare wnombre varchar(20);
declare wapellidos varchar(40);
declare wresto int;
declare wletra char(1);
declare wcadena varchar(23) default 'TRWAGMYFPDXBNJZSQVHLCKE'; -- (estas son las letras oficiales del dni encontradas en google)
declare wfin boolean default false;
declare wtodo varchar(5000) default ''; 

-- cursor con empleados sin la letra calculada
declare cursor1 cursor for
select dni, nombre, apellidos from empleados where letra_dni is null;

-- controlar el final del cursor
declare continue handler for not found set wfin = true;

-- cabecera del fichero
set wtodo = '          listado empleados\r          -------------------\r\r';

open cursor1;

-- leer el primer registro
fetch cursor1 into wdni, wnombre, wapellidos;

while wfin = false do

-- calcular resto del dni
set wresto = wdni % 23;

-- sacar la letra correspondiente
set wletra = substring(wcadena, wresto + 1, 1);

-- actualizar la letra en la tabla
update empleados set letra_dni = wletra where dni = wdni;

-- añadir datos al fichero
set wtodo = concat(wtodo, wdni, wletra, ' - ', wnombre, ' - ', wapellidos, '\r');

-- leer el siguiente registro
fetch cursor1 into wdni, wnombre, wapellidos;

end while;

close cursor1;

-- crear el fichero 
select wtodo into outfile 'C:/ProgramData/MySQL/MySQL Server 8.0/Uploads/resultado.txt';

end $$

delimiter ;
