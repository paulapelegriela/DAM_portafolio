/* TABLAS */

/* Creamos la tabla de clientes */

CREATE TABLE clientes (
nif varchar(9) primary key not null,
nom_cliente varchar(20) not null,
apellidos_cliente varchar(45) not null,
fec_nacimiento date not null,
dir_cod_provincia varchar(2) not null,
dir_calle varchar(30) not null,
dir_numero int not null,
dir_ciudad varchar(30) not null,
PPE varchar(50) not null);

/* Creamos la tabla de los telefonos de clientes */

CREATE TABLE clientes_telefonos (
nif_cliente varchar(9) not null,
tel_cliente varchar(12) not null,
PPE varchar(50) not null,
primary key (nif_cliente, tel_cliente));

/* Creamos la tabla de los proveedores */

CREATE TABLE proveedores (
cif varchar(9) primary key not null,
nombre varchar(30) not null,
dir_central_calle varchar(30) not null,
dir_central_telefono varchar(12) not null,
dir_central_ciudad varchar(30) not null,
PPE varchar(50) not null);

/* Creamos la tabla de los números de calle de los proveedores */

CREATE TABLE proveedores_num_calle_central (
cif_proveedor varchar(9) not null,
numero_calle int not null,
PPE varchar(50) not null,
primary key (cif_proveedor, numero_calle));

/* Creamos la tabla de los muebles */

CREATE TABLE muebles (
codigo int primary key not null,
descripcion varchar(200) not null,
precio double not null,
cif_proveedor varchar(9) not null,
nif_cliente varchar(9),
PPE varchar(50) not null);

/* Creamos la tabla de las provincias */

CREATE TABLE provincias (
cod_provincia varchar(2) primary key not null,
nom_provincia varchar(45) not null,
cod_comunidad varchar(2) not null,
PPE varchar(50) not null);

/* Creamos la tabla de las comunidades */

CREATE TABLE comunidades (
cod_comunidad varchar(2) primary key not null,
nom_comunidad varchar(45) not null,
PPE varchar(50) not null);

/* RELACIONES */

/* relación en la tabla clientes */

ALTER TABLE clientes
add foreign key (dir_cod_provincia) references provincias(cod_provincia);

/* relación en la tabla clientes_telefonos */

ALTER TABLE clientes_telefonos
add foreign key (nif_cliente) references clientes(nif);

/* relación en la tabla proveedores_num_calle_central */

ALTER TABLE proveedores_num_calle_central
add foreign key (cif_proveedor) references proveedores(cif);

/* relaciones en la tabla muebles */

ALTER TABLE muebles
add foreign key (cif_proveedor) references proveedores(cif);

ALTER TABLE muebles
add foreign key (nif_cliente) references clientes(nif);

/* relación en la tabla provincias */

ALTER TABLE provincias
add foreign key (cod_comunidad) references comunidades(cod_comunidad);
